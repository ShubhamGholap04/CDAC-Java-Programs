1. Write a program to read n integers into an array. Calculate and display the sum of all elements and their average.

import java.util.Scanner;

public class Average
{
    public static void main(String[] args) 
    {
        Scanner sc = new Scanner(System.in);
        
        System.out.println("Enter n:");
        int n = sc.nextInt();

        int a[] = new int[n];
        int sum = 0 ;

        System.out.println("Enter "+n+" integers:");

        for(int i=0 ; i<n ; i++)
        {
            a[i] = sc.nextInt();
            sum = sum + a[i];
        }

        double average = (double) sum/n;

        System.out.println("sum is: " + sum);
        System.out.println("average is: " + average);
        sc.close();
    }
}

7. Write a program to read a line of text and count the number of vowels and consonants present in it.

import java.util.Scanner;

public class VowelConsonant
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter a line: ");
        String s = sc.nextLine();

        int vowel = 0, consonant = 0;

        for(int i = 0; i < s.length(); i++)
        {
            char ch = s.charAt(i);

            if(ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u' ||
               ch == 'A' || ch == 'E' || ch == 'I' || ch == 'O' || ch == 'U')
            {
                vowel++;
            }
            else if((ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z'))
            {
                consonant++;
            }
        }

        System.out.println("Vowels = " + vowel);
        System.out.println("Consonants = " + consonant);
    }
}

8. Write a program to read a string and check whether it is a palindrome.

import java.util.Scanner;

public class Palindrome
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter a string: ");
        String s = sc.nextLine();

        boolean palindrome = true;

        for(int i = 0; i < s.length() / 2; i++)
        {
            if(s.charAt(i) != s.charAt(s.length() - 1 - i))
            {
                palindrome = false;
                break;
            }
        }

        if(palindrome)
            System.out.println("Palindrome");
        else
            System.out.println("Not Palindrome");
    }
}

11. Write a program to find the second largest distinct element in an array.

import java.util.Scanner;

public class SecondLargest
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();

        int a[] = new int[n];

        System.out.println("Enter elements:");

        for(int i = 0; i < a.length; i++)
        {
            a[i] = sc.nextInt();
        }

        int largest = a[0];
        int second = 0;
        boolean found = false;

        for(int i = 1; i < a.length; i++)
        {
            if(a[i] > largest)
            {
                second = largest;
                largest = a[i];
                found = true;
            }
            else if(a[i] < largest)
            {
                if(!found || a[i] > second)
                {
                    second = a[i];
                    found = true;
                }
            }
        }

        if(found)
            System.out.println("Second largest = " + second);
        else
            System.out.println("-1");
    }
}

12. Write a program to read an array and print each distinct element along with the number of times it appears.

import java.util.Scanner;

public class Frequency
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();

        int a[] = new int[n];

        System.out.println("Enter elements:");

        for(int i = 0; i < a.length; i++)
        {
            a[i] = sc.nextInt();
        }

        for(int i = 0; i < a.length; i++)
        {
            int count = 0;
            boolean alreadyPrinted = false;

            // Check if element was already printed
            for(int j = 0; j < i; j++)
            {
                if(a[i] == a[j])
                {
                    alreadyPrinted = true;
                    break;
                }
            }

            if(alreadyPrinted)
                continue;

            // Count occurrence
            for(int j = 0; j < a.length; j++)
            {
                if(a[i] == a[j])
                    count++;
            }

            System.out.println(a[i] + " = " + count);
        }
    }
}
